    @contextmanager
def set_ctx_var[T](var: ContextVar[T], value: T) -> Generator[T]:
    """Set a context variable within a context manager, restoring the previous value on exit.

    Args:
        var: The context variable to set.
        value: The value to set the context variable to.

    Returns:
        A context manager that yields the set value.
    """
    token = var.set(value)
    try:
        yield value
    finally:
        var.reset(token)

    def container_to_dict(
    container: Sequence | Set | Mapping | DataclassInstance | BaseModel,
    recursive: bool = True,
    _registry: dict[int, dict[Any, Any]] | None = None,
) -> dict[Any, Any] | None:
    """Turn an arbitrary container object into a dict.

    Args:
        container: The container to convert.
        recursive: Whether to recursively convert nested containers.
        _registry: Internal registry to track visited objects for cycle detection.

    Returns:
        A dict representation of the container, or None if not convertible.
    """
    _registry = _registry or {}

    obj_id = id(container)
    if obj_id in _registry:
        return _registry[obj_id]

    match container:
        case DataclassInstance():
            res = {
                Attr(f_name): getattr(container, f_name)
                for f_name in container.__dataclass_fields__.keys()
            }
        case _ if isinstance(container, BaseModel):
            res = {Attr(k): v for k, v in container.model_dump().items()}
        case Mapping():
            res = dict(container)
        case str() | bytes() | bytearray():
            # Scalars are iterable, but not containers to be indexed.
            return None
        case Iterable():
            res = {i: v for i, v in enumerate(container)}
        case _:
            return None

    _registry[obj_id] = res

    if recursive:
        for k, v in list(res.items()):
            m = container_to_dict(v, True, _registry)
            res[k] = m if m is not None else v

    return res

    def copy_and_override[**Params, DC: DataclassInstance](
    _init: Callable[Params, DC] | MaskedInit[Params, DC] | type[DC],
    _obj: DataclassInstance | tuple[DataclassInstance, ...],
    *args: Params.args,
    **kwargs: Params.kwargs,
) -> DC:
    """Re-construct a dataclass instance with all its init params + override.

    Args:
        _init: The dataclass constructor or a `MaskedInit` wrapping it.
        _obj: The dataclass instance(s) to copy values from.
        *args: Positional arguments to override.
        **kwargs: Keyword arguments to override.

    Returns:
        A new dataclass instance with the specified overrides.

    Note:
        Required attributes of the target dataclass type must always be provided as args / kwargs.

    Warning:
        Does not work for kw_only dataclasses and InitVars (yet).
    """
    if isinstance(_init, MaskedInit):
        _init = _init.constructor
    if not isinstance(_init, type):
        orig_init = get_origin(_init)
        assert orig_init is not None
        _init = orig_init

    constr_func = cast(Callable[..., DC], _init)
    target_fields = list(fields(cast(type, _init)))

    objs = _obj if isinstance(_obj, tuple) else (_obj,)

    obj_fields: dict[Field, Any] = {}
    for obj in objs:
        src_fields = set(f.name for f in obj.__dataclass_fields__.values())
        all_obj_fields = {f: getattr(obj, f.name) for f in target_fields if f.name in src_fields}
        obj_fields = {
            f: (
                v
                if v is not MISSING
                else obj_fields.get(
                    f,
                    (
                        f.default
                        if f.default is not MISSING
                        else (f.default_factory() if f.default_factory is not MISSING else MISSING)
                    ),
                )
            )
            for f, v in all_obj_fields.items()
        }

    assert not any(v is MISSING for v in obj_fields.values())

    obj_kwargs = list({f.name: v for f, v in obj_fields.items() if f.init}.items())
    new_kwargs = {
        **dict(obj_kwargs[len(args) :] if len(args) < len(obj_kwargs) else {}),
        **kwargs,
    }

    new_obj = constr_func(*args, **new_kwargs)  # pyright: ignore[reportCallIssue]

    for f, v in obj_fields.items():
        if not f.init:
            setattr(new_obj, f.name, v)

    return new_obj

    def flatten_dict(
    d: Mapping[Any, Any],
    path: tuple = (),
    _registry: dict[int, dict[tuple, Any]] | None = None,
) -> dict[tuple, Any]:
    """Turn a nested mapping into a flat dict (ignoring recursive references).

    Args:
        d: The mapping to flatten.
        path: The current path of keys (used for recursion).
        _registry: Internal registry to track visited objects for cycle detection.

    Returns:
        A flat dict mapping tuple paths to values.
    """
    items: dict[tuple, Any] = {}

    _registry = _registry or {}

    obj_id = id(d)
    if obj_id in _registry:
        return _registry[obj_id]

    _registry[obj_id] = items

    for k, v in d.items():
        new_path = path + (k,)
        if isinstance(v, Mapping):
            items.update(flatten_dict(v, new_path, _registry))
        else:
            items[new_path] = v

    return items

    def get_module_distribution(module: ModuleType) -> meta.Distribution | None:
    """Get the distribution package of given module, if any.

    Args:
        module: Module to inspect.

    Returns:
        Distribution package of the module, or ``None`` if not part of a distribution.
    """
    mod_file = get_module_file(module)
    if mod_file is None:
        return None

    dists = {
        Path(str(dist.locate_file(f"{name}"))): dist for name, dist in get_distributions().items()
    }

    # An editable install resolves to the repository root, which contains the
    # whole virtual environment; the most specific (deepest) match wins so a
    # module in ``site-packages`` is attributed to its own distribution.
    mod_dists: list[tuple[int, meta.Distribution]] = []
    for dist_path, dist in dists.items():
        if mod_file.is_relative_to(dist_path):
            mod_dists.append((len(dist_path.parts), dist))
        elif dist.origin is not None and "url" in dist.origin.__dict__:  # type: ignore
            url = str(dist.origin.url)
            if url.startswith("file://"):
                path = _file_url_to_path(url)
                if mod_file.is_relative_to(path):
                    mod_dists.append((len(path.parts), dist))

    if len(mod_dists) == 0:
        return None

    return max(mod_dists, key=lambda item: item[0])[1]

def gen_int_hash(_ctx: set[int], obj: Any) -> int:  # noqa: C901
    """Generate stable hash for obj (must be known, hashable or composed of such).

    Args:
        obj: Object to hash.

    Returns:
        Hash of the object as integer.
    """
    match obj:
        case None:
            return 0
        case ModuleType():
            return gen_int_hash(obj.__name__)
        case type() | TypeAliasType():
            return gen_int_hash((obj.__module__, obj.__name__))
        case FunctionType() if hasattr(obj, "__name__"):
            return gen_int_hash(str(PyObjectRef.reference(obj).url))
        case FunctionType():
            # This may be unstable as closures are not considered.
            return gen_int_hash(list(obj.__code__.co_lines()))
        case GenericAlias() | UnionType():
            base = get_origin(obj)
            args = get_args(obj)
            return gen_int_hash(
                (
                    (
                        "types.UnionType"
                        if base is UnionType
                        else str(PyObjectRef.reference(base).url)
                    ),
                    *(gen_int_hash(arg) for arg in args),
                )
            )
        case Number() | str() | bytes() | date() | time() | datetime() | timedelta():
            return _stable_hash(obj)
        case tuple():
            return _hash_sequence(obj, _ctx)
        case _ if id(obj) in _ctx:
            return 0
        case _ if isinstance(obj, pd.DataFrame | pd.Series | pd.Index):
            # pandas types ``hash_pandas_object`` loosely, hence the ignore.
            return sum(hash_pandas_object(obj))  # pyright: ignore[reportCallIssue]
        case list():
            _ctx.add(id(obj))
            return _hash_sequence(obj, _ctx)
        case dict():
            _ctx.add(id(obj))
            return sum(_hash_sequence(item, _ctx) for item in obj.items())
        case set():
            _ctx.add(id(obj))
            return sum(gen_int_hash(item) for item in obj)
        case Hashable():
            _ctx.add(id(obj))
            return hash(obj)
        case DataclassInstance():
            _ctx.add(id(obj))
            return gen_int_hash({f.name: getattr(obj, f.name) for f in fields(obj)})
        # ``object`` has provided ``__getstate__`` since Python 3.11, so the
        # protocol alone matches every instance; require a custom state hook.
        case StatePicklable() if type(obj).__getstate__ is not object.__getstate__:
            state = obj.__getstate__()
            state = state if isinstance(state, tuple) else (state,)
            state_dicts = [s for s in state if isinstance(s, dict)]

            if len(state_dicts) == 0:
                raise ValueError()

            return sum(sum(_hash_sequence((k, v), _ctx) for k, v in s.items()) for s in state_dicts)
        case ArgsPicklable():
            args = obj.__getnewargs__()
            return sum(gen_int_hash(item) for item in args)
        case ArgsPicklableEx():
            args, kwargs = obj.__getnewargs_ex__()
            return sum(gen_int_hash(item) for item in args) + sum(
                gen_int_hash(item) for item in kwargs.items()
            )
        case _:
            if pl is not None:
                match obj:
                    case pl.DataFrame():
                        return sum(obj.hash_rows()) if not obj.is_empty() else 0
                    case pl.Series():
                        return sum(obj.hash())
                    case _:
                        pass

            raise ValueError()

    """Create a reference from given URL.

        # so they must not leak into the repository URL.

        version=url.query.get("version"),
        py_version=url.query.get("python"),
        module_name=module,
        qualname=object,

def reference(  # noqa: C901
    obj: T2,
    version: str | None = None,
    version_strategy: VersionStrategy = "major",
) -> "PyObjectRef[T2]":
    """Create a reference to given object.

    Args:
        obj: Object to create reference for.
        version: Exact or range version specifier for the object's package/repo.
        version_strategy: Strategy to use for version range specification.

    Returns:
        Reference to the given object.
    """
    qualname = getattr(obj, "__qualname__", None)
    if qualname is None:
        raise ValueError("Object must have fully qualified name (`__qualname__`)")

    module = inspect.getmodule(obj)
    if module is None:
        raise ValueError("Object must be associated with a module.")

    dist = get_module_distribution(module)
    package = dist.name if dist is not None else None

    git_repo = None
    repo = None

    if dist is not None and dist.origin is not None:
        # Fetch the origin URL of the distribution, if any.
        repo = dict(dist.origin.__dict__).get("url")

    if dist is None or str(repo).startswith("file://"):
        # Not part of distribution or local file origin URL
        # -> assume first-party code commited to Git repo.
        git_repo = get_module_repo(module)
        if git_repo is not None:
            repo = git_repo.remote().url

    if repo is None:
        if dist is not None:
            # Code is part of distribution, but no explicit origin URL
            # -> assume PyPI indexed package
            repo = f"https://pypi.org/simple/{package}"
        elif module.__name__ != "builtins":
            # No distribution, no Git repo and not built-in module
            raise ValueError("Non-builtin object must be associated with a repository.")

    version_exact = None
    semver_exact = None

    if version is None:
        if dist is not None:
            version_exact = dist.version
            try:
                semver_exact = Version(version_exact)
            except InvalidVersion:
                pass
        elif git_repo is not None:
            current_tags = [tag for tag in git_repo.tags if tag.commit == git_repo.head.commit]
            if len(current_tags) > 0:
                for tag in current_tags:
                    version_exact = tag.name
                    try:
                        semver_exact = Version(version_exact)
                        break
                    except InvalidVersion:
                        continue
            elif not git_repo.head.is_detached:
                version_exact = git_repo.active_branch.name
                try:
                    semver_exact = Version(version_exact)
                except InvalidVersion:
                    pass

    if semver_exact is not None:
        version = version_to_range(semver_exact, version_strategy)
    elif version_strategy == "exact" and version_exact is not None:
        version = f"=={version_exact}"

    return PyObjectRef(
        object_type=type(obj),
        qualname=qualname,
        module_name=module.__name__,
        repo_url=repo,
        version=version,
        py_version=version_to_range(Version(f"{platform.python_version()}"), "minor"),
        _module=module,
        _dist=dist,
        _object=obj,
    )

    """

        _ext_deps = set()

        # the entry point as one of its own dependencies.
        _int_deps = {module}

    deps = [dep for _, m in inspect.getmembers(module) if (dep := inspect.getmodule(m)) is not None]
    ext_deps_map = {
        dep: dist.metadata["Name"]
        for dep in deps
        if (dist := get_module_distribution(dep)) is not None and dist.metadata is not None
    }
    new_ext_deps = set(ext_deps_map.values()) - _ext_deps

    these_int_deps = set(deps) - set(ext_deps_map.keys())
    new_int_deps = these_int_deps - _int_deps

    if len(new_ext_deps) > 0 or len(new_int_deps) > 0:
        _ext_deps |= new_ext_deps
        _int_deps |= new_int_deps

        for dependency in new_int_deps:
            sub_ext_deps, sub_int_deps = get_all_module_dependencies(
                dependency, _ext_deps, _int_deps
            )
            _ext_deps |= sub_ext_deps
            _int_deps |= sub_int_deps

        return _ext_deps, _int_deps - {module}

    return _ext_deps, _int_deps - {module}

    def semver_range_to_spec(semver_range: str) -> SpecifierSet:
    """Convert semver range to Python version specifier.

    Args:
        semver_range: Semver range string.

    Returns:
        Corresponding Python version specifier set.
    """
    match = re.match(r"^(?P<op>\^|~|>=|<=|==|=|>|<)?(?P<version>.+)$", semver_range.strip())
    if match is None:
        raise ValueError(f"Invalid semver range: '{semver_range}'.")

    op = match.group("op")
    version = Version(match.group("version").strip())

    if op == "^":
        return SpecifierSet(f">={version.public},<{version.major + 1}")

    if op == "~":
        return SpecifierSet(f">={version.public},<{version.major}.{version.minor + 1}")

    if op in (">=", "<=", ">", "<"):
        return SpecifierSet(f"{op}{version.public}")

    return SpecifierSet(f"=={version.public}")

    def version_diff(v1: Version, v2: Version) -> Version | None:
    """Get the difference between two versions (v1 - v2).

    Args:
        v1: Newer version.

        Dictionary of all args + kwargs.
    """
    argspec = inspect.getfullargspec(func)

    posonlyargs = list(getattr(argspec, "posonlyargs", ()))
    posarg_names = [*posonlyargs, *argspec.args]

    arg_defaults = argspec.defaults or []
    defaulted_names = posarg_names[len(posarg_names) - len(arg_defaults) :]
    defaults = dict(zip(defaulted_names, arg_defaults))
    defaults |= dict(argspec.kwonlydefaults or {})

    # Handle bound methods
    if len(posarg_names) > 0 and posarg_names[0] == "self" and hasattr(func, "__self__"):
        posarg_names = posarg_names[1:]

    posargs = dict(zip(posarg_names[: len(args)], args))

    return {**defaults, **posargs, **(kwargs or {})}

    def get_inheritance_distance(cls: type, base: type) -> int | None:
    """Return the inheritance distance between two classes.

    Note: Positive direction is from subclass to base class. If arguments are
    are reversed, the sign will be negative.

    Warning: Untested function.

    Args:
        cls: The subclass.
        base: The base class.

    Returns:
        The signed inheritance distance between the two classes. If the base class is
        not a base of the subclass, None is returned.
    """
    if not isinstance(cls, type) or not isinstance(base, type):
        return None

    if cls is base:
        return 0

    if issubclass(cls, base):
        cls, base, sign = (cls, base, 1)
    elif issubclass(base, cls):
        cls, base, sign = (base, cls, -1)
    else:
        return None

    distance = 1
    bases = set(cls.__bases__)
    while base not in bases and distance < 100:
        bases = reduce(set.union, (set(b.__bases__) for b in bases))
        distance += 1

    return distance * sign

    """Return the lowest common base of given types.

    def set_typeargs[T](  # noqa: C901
    typedef: TypeForm[T],
    args: Sequence[AnyTypeHint] | dict[TypeVar, AnyTypeHint],
) -> GenericAlias[T]:
    """Set a typevar in a generic type hint.

    Args:
        typedef: The type definition to set type arguments for.
        args: The type arguments to set, either positionally or by type variable.

    Returns:
        The generic type with set type arguments.

    Raises:
        TypeError: If ``typedef`` is not a parameterized generic type.
        ValueError: If ``args`` names a type variable that ``typedef`` does not use.
    """
    is_type_aware = isinstance(typedef, type) and issubclass(typedef, TypeAwareClass)

    if is_type_aware:
        orig = getattr(typedef, "__origin__", None)
        current_args = getattr(typedef, "__args__", ())
    else:
        orig = get_origin(typedef)
        current_args = get_args(typedef)

    if orig is None:
        raise TypeError(f"`{typedef!r}` is not a generic type definition.")

    if len(current_args) == 0:
        raise TypeError(f"`{typedef!r}` has no type arguments to replace.")

    def _apply(values: tuple[Any, ...]) -> GenericAlias[T]:
        if is_type_aware:
            # A starred subscript is delivered as a single tuple key, so a
            # single argument must be passed without the star.
            return cast(GenericAlias[T], orig[values[0] if len(values) == 1 else values])

        return cast(GenericAlias[T], orig[*values])

    if not isinstance(args, dict):
        arg_seq = tuple(args)
        if len(arg_seq) == 0:
            raise ValueError("At least one type argument is required.")

        return _apply(arg_seq)

    params = getattr(orig, "__parameters__", None) or getattr(orig, "__type_params__", ())
    if not isinstance(params, tuple) or len(params) == 0:
        params = tuple(
            p
            for p in current_args
            if isinstance(p, TypeVar | ExtTypeVar | TypeVarTuple | ExtTypeVarTuple)
        )

    # Map each parameter to the argument that currently fills it.
    typearg_map = dict(zip(params, current_args))
    typevar_map = TypeRef(typedef).typevar_map

    for typevar, arg in args.items():
        if typevar in typearg_map:
            typearg_map[typevar] = arg
            continue

        targets = [p for p, supplied in typearg_map.items() if supplied is typevar]

        if len(targets) == 0:
            # Follow a substitution chain, e.g. ``Base[T]`` inside ``Sub[T]``.
            subs = typevar_map.get(typevar)
            resolved = subs.typedef if isinstance(subs, TypeRef) else None
            if isinstance(resolved, TypeVar | ExtTypeVar):
                targets = [p for p, supplied in typearg_map.items() if supplied is resolved]

        if len(targets) == 0:
            raise ValueError(f"Type variable `{typevar!r}` is not a parameter of `{typedef!r}`.")

        for target in targets:
            typearg_map[target] = arg

    return _apply(tuple(typearg_map.values()))

    def typedef_to_typeset(  # noqa: C901
    typedef: AnyTypeHint | None,
    typevar_map: (
        dict[
            TypeVar | ExtTypeVar | TypeVarTuple | ExtTypeVarTuple,
            TypeRef | tuple[TypeRef, ...],
        ]
        | None
    ) = None,
    ctx_module: ModuleType | None = None,
    remove_null: bool = False,
) -> set[type]:
    """Convert type definition to set of types (>1 in case of union).

    Args:
        typedef: The type definition to convert.
        typevar_map: Mapping of type variables to their actual types.
        ctx_module: Context module for resolving forward-referenced types.
        remove_null: Whether to remove ``NoneType`` from the resulting typeset.

    Returns:
        Set of types.
    """
    typedef_set: set[AnyTypeHint | None] = {typedef}
    root_orig = get_origin(typedef)

    if isinstance(typedef, UnionType) or root_orig is Union:
        typedef_set = {get_origin(union_arg) or union_arg for union_arg in get_args(typedef)}

        if remove_null:
            typedef_set &= {t for t in typedef_set if t is not None and not is_subtype(t, NoneType)}

    typeset: set[type] = set()

    for t in typedef_set:
        if t is None:
            typeset.add(NoneType)
            continue

        if isinstance(t, type):
            typeset.add(t)
            continue

        if isinstance(t, TypeAliasType):
            # A type alias is a synonym, so its typeset is the value's typeset.
            typeset |= typedef_to_typeset(
                t.__value__,
                typevar_map=typevar_map,
                ctx_module=ctx_module,
                remove_null=remove_null,
            )
            continue

        if isinstance(t, NewType):
            supertype = t.__supertype__
            typeset.add(supertype if isinstance(supertype, type) else object)
            continue

        if isinstance(t, TypeVar | ExtTypeVar):
            # An unresolved type variable carries no concrete type.
            typeset.add(object)
            continue

        t_parsed = t

        orig = get_origin(t_parsed)

        if isinstance(orig, TypeAliasType):
            # A parameterized generic alias, e.g. ``MyList[int]`` where
            # ``type MyList[T] = list[T]``; resolve the alias' value.
            typeset |= typedef_to_typeset(
                TypeRef(
                    orig.__value__,
                    subs=TypeRef(cast(Any, t)).local_typevar_map,
                    ctx_module=ctx_module,
                ).typeform,
                typevar_map=typevar_map,
                ctx_module=ctx_module,
                remove_null=remove_null,
            )
            continue

        if orig is Annotated:
            typeset |= typedef_to_typeset(
                get_args(t)[0],
                typevar_map=typevar_map,
                ctx_module=ctx_module,
                remove_null=remove_null,
            )
            continue

        if orig is None or orig is Literal:
            typeset.add(object)
            continue

        if orig is UnionType or orig is Union:
            typeset |= typedef_to_typeset(
                t_parsed,
                typevar_map=typevar_map,
                ctx_module=ctx_module,
                remove_null=remove_null,
            )
            continue

        typeset.add(orig if isinstance(orig, type) else object)

    return typeset
